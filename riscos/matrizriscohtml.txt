  <script>
    grist.ready({ requiredAccess: 'read table' });
    
    let allRecords = [];
    let selectedDepartments = []; // store selected departments
    let allDepartments = []; // store all available departments
    let debugMode = false; // Set to true to enable debug info
    
    // Helper function to get the $PAs value.
    function getPAValue(r) {
      // Check for all possible property names
      let value = null;
      
      // Check for property names in different case combinations
      const possibleProps = ["$PAs", "PAs", "$pas", "pas", "$PA", "PA", "$pa", "pa"];
      
      for (const prop of possibleProps) {
        if (r[prop] !== undefined) {
          value = r[prop];
          break;
        }
      }
      
      // Log the found value for debugging
      if (debugMode) {
        console.log(`PA value for record ${r.id || 'unknown'}: ${value}`);
      }
      
      return Number(value) || 0;
    }
    
    // Helper function to get the overdue value (Atrasado_).
    function getOverdueValue(r) {
      // Check for all possible property names
      let value = null;
      
      // Check for property names in different case combinations
      const possibleProps = ["$Atrasado_", "Atrasado_", "$atrasado_", "atrasado_"];
      
      for (const prop of possibleProps) {
        if (r[prop] !== undefined) {
          value = r[prop];
          break;
        }
      }
      
      return Number(value) || 0;
    }
    
    // Helper function to get the latest Tratamento (Aceitar, Mitigar, Eliminar)
    function getTratamento(r) {
      // Check for various possible property names
      const possibleProps = ["$Ultimo_Tratamento", "Ultimo_Tratamento", "$ultimo_tratamento", "ultimo_tratamento", "Tratamento", "$Tratamento"];
      
      for (const prop of possibleProps) {
        if (r[prop] !== undefined && r[prop] !== "") {
          if (debugMode) {
            console.log(`Tratamento for record ${r.id || 'unknown'}: ${r[prop]}`);
          }
          return r[prop];
        }
      }
      
      return ""; // Default to empty if not specified
    }
    
    // Helper function to check if a risk requires an action plan
    function requiresActionPlan(risk) {
      const tratamento = getTratamento(risk);
      const paValue = getPAValue(risk);
      
      // Debug logs
      if (debugMode) {
        console.log(`Risk ID: ${risk.id || 'unknown'}, Tratamento: ${tratamento}, PAs: ${paValue}`);
      }
      
      // Check both normalized and original case for comparison
      const normalizedTratamento = tratamento.toLowerCase();
      return (normalizedTratamento === "mitigar" || 
              normalizedTratamento === "eliminar" || 
              tratamento === "Mitigar" || 
              tratamento === "Eliminar") && 
             paValue === 0;
    }
    
    grist.onRecords(function(records) {
      console.log("Records loaded:", records); // Debugging line
      if (!records || records.length === 0) {
        document.getElementById("matrix-container").innerHTML = "No risk data available.";
        return;
      }
      allRecords = records;
      
      if (debugMode) {
        // Display sample record structure
        const sampleRecord = records[0];
        const debugInfo = document.getElementById("debug-info");
        debugInfo.style.display = "block";
        debugInfo.innerHTML = "Sample record structure:\n" + JSON.stringify(sampleRecord, null, 2);
        
        // Log more detailed info about potential issues
        let risksMissingPlans = 0;
        let overdueRisks = 0;
        
        records.forEach(r => {
          const needsPlan = requiresActionPlan(r);
          const isOverdue = getOverdueValue(r) > 0;
          
          if (needsPlan) risksMissingPlans++;
          if (isOverdue) overdueRisks++;
        });
        
        console.log(`Summary: ${records.length} total risks, ${risksMissingPlans} missing plans, ${overdueRisks} overdue`);
      }
      
      createDepartmentDropdown(allRecords);
      renderMatrix(allRecords);
    });
    
    // Build the department dropdown options.
    function createDepartmentDropdown(records) {
      const dropdownOptions = document.getElementById("department-dropdown-options");
      dropdownOptions.innerHTML = "";
      const deptSet = new Set();
      records.forEach(r => {
        let dept = "SEM DEPARTAMENTO";
        if (r.Departamento) {
          if (typeof r.Departamento === "object" && r.Departamento.Departamento) {
            dept = r.Departamento.Departamento.toUpperCase();
          } else {
            dept = r.Departamento.toString().toUpperCase();
          }
        }
        deptSet.add(dept);
      });
      allDepartments = Array.from(deptSet).sort();
      updateDropdownOptions();
    }
    
    // Update the dropdown options based on selected departments.
    function updateDropdownOptions() {
      const dropdownOptions = document.getElementById("department-dropdown-options");
      dropdownOptions.innerHTML = "";
      allDepartments.forEach(dept => {
        if (!selectedDepartments.includes(dept)) {
          const label = document.createElement("label");
          const checkbox = document.createElement("input");
          checkbox.type = "checkbox";
          checkbox.value = dept;
          checkbox.addEventListener("change", function(e) {
            e.stopPropagation(); // Prevent closing the dropdown when clicking a checkbox.
            if (this.checked) {
              selectedDepartments.push(dept);
            } else {
              selectedDepartments = selectedDepartments.filter(d => d !== dept);
            }
            updateDropdownButton();
            updateDropdownOptions();
            renderMatrix(allRecords);
          });
          label.appendChild(checkbox);
          label.appendChild(document.createTextNode(" " + dept));
          dropdownOptions.appendChild(label);
        }
      });
    }
    
    // Update the dropdown button with selected departments as cards.
    function updateDropdownButton() {
      const dropdownBtn = document.getElementById("department-dropdown-btn");
      dropdownBtn.innerHTML = selectedDepartments.length > 0 ? 
        selectedDepartments.map(dept => `
          <div class="department-card">
            ${dept}
            <span class="remove-btn" onclick="removeDepartment('${dept}')">×</span>
          </div>
        `).join("") : 
        "<span>Selecione departamentos...</span>";
    }
    
    // Remove a department from the selection.
    function removeDepartment(dept) {
      selectedDepartments = selectedDepartments.filter(d => d !== dept);
      updateDropdownButton();
      updateDropdownOptions();
      renderMatrix(allRecords);
    }
    
    // Select all departments.
    function selectAllDepartments() {
      selectedDepartments = [...allDepartments];
      updateDropdownButton();
      updateDropdownOptions();
      renderMatrix(allRecords);
    }
    
    // Remove all departments.
    function removeAllDepartments() {
      selectedDepartments = [];
      updateDropdownButton();
      updateDropdownOptions();
      renderMatrix(allRecords);
    }
    
    // Attach click event to the department selection button.
    const dropdownBtn = document.getElementById("department-dropdown-btn");
    dropdownBtn.addEventListener("click", function(e) {
      const options = document.getElementById("department-dropdown-options");
      options.style.display = options.style.display === "block" ? "none" : "block";
      e.stopPropagation();
    });
    
    // Close the dropdown if clicking outside of it.
    document.addEventListener("click", function(e) {
      if (!document.getElementById("department-dropdown").contains(e.target)) {
        document.getElementById("department-dropdown-options").style.display = "none";
      }
    });
    
    // Render the risk matrix.
    function renderMatrix(records) {
      console.log("Filtered records:", records); // Debugging line
      let filteredRecords = records.filter(r => {
        let dept = "SEM DEPARTAMENTO";
        if (r.Departamento) {
          if (typeof r.Departamento === "object" && r.Departamento.Departamento) {
            dept = r.Departamento.Departamento.toUpperCase();
          } else {
            dept = r.Departamento.toString().toUpperCase();
          }
        }
        return (selectedDepartments.length === 0) || selectedDepartments.includes(dept);
      });

      let riskMap = {};
      filteredRecords.forEach(record => {
        let impacto = record.ultimpac || 0;
        let probabilidade = record.ultprob || 0;
        let key = `${probabilidade}-${impacto}`;
        if (!riskMap[key]) { riskMap[key] = []; }
        riskMap[key].push(record);
      });

      console.log("Risk map:", riskMap); // Debugging line

      let max_valor = 5;
      let html = `
        <table>
          <tr>
            <th rowspan="2" colspan="2" class="header" style="width: 20%;">⚠️ ${filteredRecords.length}<br>📄 ${filteredRecords.filter(r => r.ultimpac == 0 || r.ultprob == 0).length}</th>
            <th colspan="5" class="header">IMPACTO</th>
          </tr>
          <tr>
            ${Array.from({length: max_valor}, (_, i) => `<th class="header">${i + 1}</th>`).join('')}
          </tr>
      `;

      for (let i = max_valor; i >= 1; i--) {
        html += `<tr>`;
        if (i === max_valor) {
          html += `<th rowspan="5" class="header probabilidade-header">PROBABILIDADE</th>`;
        }
        html += `<th class="header">${i}</th>`;
        
        for (let j = 1; j <= max_valor; j++) {
          let key = `${i}-${j}`; // FIXED: Swapped i and j to match probability-impact
          let riskScore = i * j;
          let [cor_celula, nivel_risco, cor_borda] = obterCorRisco(riskScore);
          let risks = riskMap[key] || [];
          let count = risks.length;
          
          // Count risks that require action plans
          let risksWithoutPlans = risks.filter(r => requiresActionPlan(r)).length;
          let overdueRisks = risks.filter(r => getOverdueValue(r) > 0).length;
          
          // Debug logging for this cell
          if (debugMode && (risksWithoutPlans > 0 || overdueRisks > 0)) {
            console.log(`Cell ${key} (Score ${riskScore}): ${count} risks, ${risksWithoutPlans} without plans, ${overdueRisks} overdue`);
          }
          
          // Fade out the color if there are no risks.
          let fadedColor = count === 0 ? applyFade(cor_celula) : cor_celula;
          
          // Build tooltip text.
          let tooltipLines = [];
          if (risks.length > 0) {
            let risksByDept = {};
            risks.forEach(r => {
              let dept = "SEM DEPARTAMENTO";
              if (r.Departamento) {
                if (typeof r.Departamento === "object" && r.Departamento.Departamento) {
                  dept = r.Departamento.Departamento.toUpperCase();
                } else {
                  dept = r.Departamento.toString().toUpperCase();
                }
              }
              if (!risksByDept[dept]) { risksByDept[dept] = []; }
              risksByDept[dept].push(r);
            });
            let depts = Object.keys(risksByDept).sort();
            depts.forEach((dept) => {
              if (tooltipLines.length > 0) tooltipLines.push("");
              tooltipLines.push(`🔶 ${dept}`);
              risksByDept[dept].sort((a, b) => a.NomeRisco.localeCompare(b.NomeRisco))
                .forEach(r => {
                  let riskId = r.id;
                  let formattedId = ("0000" + riskId).slice(-4);
                  let riskLine = "  RSK-" + formattedId;
                  if (requiresActionPlan(r)) riskLine += " 🔴";
                  if (getOverdueValue(r) > 0) riskLine += " 🟡";
                  riskLine += " : " + r.NomeRisco;
                  tooltipLines.push(riskLine);
                });
            });
          } else {
            tooltipLines.push("Nenhum risco nesta célula");
          }
          let tooltipText = tooltipLines.join("&#013;").replace(/"/g, '&quot;');
          
          html += `<td class="risk-cell" title="${tooltipText}" style="background-color: ${fadedColor}; --border-color: ${cor_borda};">
            <div class="counter">${count}</div>
            ${risksWithoutPlans > 0 ? `<div class="alert-badge">${risksWithoutPlans}</div>` : ''}
            ${overdueRisks > 0 ? `<div class="overdue-badge">${overdueRisks}</div>` : ''}
          </td>`;
        }
        html += `</tr>`;
      }
      
      html += `</table>`;
      html += `<button class="legend-button" title="Registro de Riscos&#013;⚠️ Total de Riscos&#013;📄 Somente Documentados&#013;Células da Matriz&#013;⚪ Riscos na célula&#013;🔴 Riscos sem planos de ação&#013;🟠 Riscos com prazo ultrapassado&#013;Graus de Risco&#013;🟦 Muito Baixo&#013;🟩 Baixo&#013;🟨 Moderado&#013;🟧 Elevado&#013;🟥 Extremo">Legenda</button>`;
      
      // Add debug toggle button if debug mode is enabled
      if (debugMode) {
        html += `<button onclick="document.getElementById('debug-info').style.display = document.getElementById('debug-info').style.display === 'none' ? 'block' : 'none';" style="margin-left: 10px; background-color: #555;">Debug Info</button>`;
      }
      
      document.getElementById("matrix-container").innerHTML = html;
    }
    
    // Helper function to fade out a color.
    function applyFade(color) {
      let opacity = 0.3; // Adjust opacity for faded effect.
      return `rgba(${parseInt(color.slice(1, 3), 16)}, ${parseInt(color.slice(3, 5), 16)}, ${parseInt(color.slice(5, 7), 16)}, ${opacity})`;
    }
    
    function obterCorRisco(score) {
      if (score === 0) return ['#d3d3d3', 'Inexistente', '#a9a9a9'];
      if (score <= 2) return ['#157AFB', 'Muito Baixo', '#0e5cad'];
      if (score <= 5) return ['#2AE028', 'Baixo', '#1f9d1f'];
      if (score <= 10) return ['#E8D62F', 'Moderado', '#b39b1e'];
      if (score <= 16) return ['#FD9D28', 'Elevado', '#c7761c'];
      return ['#E00A17', 'Extremo', '#a10b12'];
    }
