    <script src="https://docs.getgrist.com/grist-plugin-api.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/chart.js"></script>

    <script>
        let allRecords = [];
        let mappings = {};
        let chartInstances = {}; // Um objeto para guardar todas as instâncias de gráficos
        const mainContainer = document.querySelector('.container');

        grist.ready({
            columns: [
                { name: "obraColumn", title: "Coluna de Obra" },
                { name: "emitenteColumn", title: "Coluna de Fornecedor" },
                { name: "valorColumn", title: "Coluna de Valor" },
                { name: "inspecaoColumn", title: "Coluna de Inspeção" }, // Novo mapeamento
            ],
            requiredAccess: 'full',
        });

        grist.onRecords(function(records, newMappings) {
            allRecords = records;
            mappings = newMappings;
            if (!mappings.obraColumn || !mappings.emitenteColumn || !mappings.valorColumn || !mappings.inspecaoColumn) {
                document.getElementById('loading').style.display = 'none';
                document.getElementById('config-prompt').style.display = 'block';
                mainContainer.style.display = 'none';
                return;
            }
            document.getElementById('loading').style.display = 'none';
            document.getElementById('config-prompt').style.display = 'none';
            mainContainer.style.display = 'flex';
            main();
        });

        function main() {
            populateFilter(allRecords);
            addEventListeners();
            updateDisplay();
            renderAllCharts(allRecords);
        }

        function addEventListeners() {
            document.getElementById('obraFilter').addEventListener('change', () => {
                const selectedObra = document.getElementById('obraFilter').value;
                const filteredData = selectedObra === 'all' ? allRecords : allRecords.filter(r => r[mappings.obraColumn] === selectedObra);
                renderAllCharts(filteredData);
            });
            document.querySelectorAll('input[name="analysisType"], input[name="performanceView"], input[name="qualityView"]').forEach(radio => {
                radio.addEventListener('change', updateDisplay);
            });
        }

        function updateDisplay() {
            const analysisType = document.querySelector('input[name="analysisType"]:checked').value;
            
            // Controla a visibilidade dos seletores e dos containers de gráficos
            mainContainer.classList.toggle('show-performance', analysisType === 'performance');
            mainContainer.classList.toggle('show-quality', analysisType === 'quality');

            if (analysisType === 'performance') {
                const perfView = document.querySelector('input[name="performanceView"]:checked').value;
                const perfCharts = document.querySelector('.performance-charts');
                perfCharts.className = 'performance-charts'; // Reset
                if (perfView !== 'both') perfCharts.classList.add('view-single', `view-${perfView}`);
            } else { // analysisType === 'quality'
                const qualityView = document.querySelector('input[name="qualityView"]:checked').value;
                const qualityCharts = document.querySelector('.quality-charts');
                qualityCharts.className = 'quality-charts'; // Reset
                qualityCharts.classList.add(`view-${qualityView}`);
            }

            // Redimensiona todos os gráficos para se ajustarem ao novo layout
            setTimeout(() => {
                Object.values(chartInstances).forEach(chart => chart && chart.resize());
            }, 50);
        }

        function populateFilter(records) {
            const obraColumn = mappings.obraColumn;
            const obras = new Set(records.map(r => r[obraColumn]));
            const filterSelect = document.getElementById('obraFilter');
            const selectedValue = filterSelect.value;
            filterSelect.innerHTML = '<option value="all">Todas as Obras</option>';
            Array.from(obras).sort().forEach(obra => {
                if (obra) filterSelect.innerHTML += `<option value="${obra}">${obra}</option>`;
            });
            filterSelect.value = selectedValue && Array.from(filterSelect.options).some(o => o.value === selectedValue) ? selectedValue : 'all';
        }
        
        function renderAllCharts(records) {
            // -- Análise de Performance --
            const perfSummary = records.reduce((acc, r) => {
                const emitente = r[mappings.emitenteColumn];
                if (!emitente) return acc;
                if (!acc[emitente]) acc[emitente] = { count: 0, totalValue: 0 };
                acc[emitente].count++;
                acc[emitente].totalValue += parseFloat(r[mappings.valorColumn]) || 0;
                return acc;
            }, {});
            const perfArray = Object.keys(perfSummary).map(k => ({ emitente: k, ...perfSummary[k] }));
            const sortedByCount = [...perfArray].sort((a, b) => b.count - a.count).slice(0, 15);
            const sortedByValue = [...perfArray].sort((a, b) => b.totalValue - a.totalValue).slice(0, 15);
            drawBarChart('entregasChart', sortedByCount.map(d => d.emitente), sortedByCount.map(d => d.count), 'Top 15 Fornecedores por Nº de Entregas', 'number');
            drawBarChart('valorChart', sortedByValue.map(d => d.emitente), sortedByValue.map(d => d.totalValue), 'Top 15 Fornecedores por Valor Total', 'currency');

            // -- Análise de Qualidade --
            // 1. % Aceitação por Obra
            const acceptanceSummary = records.reduce((acc, r) => {
                const obra = r[mappings.obraColumn];
                if (!obra) return acc;
                if (!acc[obra]) acc[obra] = { total: 0, accepted: 0 };
                acc[obra].total++;
                if (r[mappings.inspecaoColumn] === 'A') acc[obra].accepted++;
                return acc;
            }, {});
            const acceptanceArray = Object.keys(acceptanceSummary).map(k => ({
                obra: k,
                percentage: (acceptanceSummary[k].accepted / acceptanceSummary[k].total) * 100 || 0
            })).sort((a,b) => a.percentage - b.percentage);
            drawBarChart('acceptanceChart', acceptanceArray.map(d => d.obra), acceptanceArray.map(d => d.percentage), '% de Entregas Aceitas ("A") por Obra', 'percent', true);

            // 2. Problemas por Fornecedor
            const problemRecords = records.filter(r => r[mappings.inspecaoColumn] !== 'A' && r[mappings.emitenteColumn]);
            const problemSummary = problemRecords.reduce((acc, r) => {
                const emitente = r[mappings.emitenteColumn];
                acc[emitente] = (acc[emitente] || 0) + 1;
                return acc;
            }, {});
            const problemArray = Object.keys(problemSummary).map(k => ({ emitente: k, count: problemSummary[k] })).sort((a, b) => b.count - a.count).slice(0, 15);
            drawBarChart('problemsChart', problemArray.map(d => d.emitente), problemArray.map(d => d.count), 'Top 15 Fornecedores por Nº de Não Conformidades', 'number');
        }

        function drawBarChart(canvasId, labels, data, title, format, horizontal = false) {
            const ctx = document.getElementById(canvasId).getContext('2d');
            const chartConfig = {
                type: 'bar',
                data: { labels, datasets: [{ label: title, data, backgroundColor: 'rgba(54, 162, 235, 0.6)', borderColor: 'rgba(54, 162, 235, 1)', borderWidth: 1 }] },
                options: {
                    indexAxis: horizontal ? 'y' : 'x',
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { display: false },
                        title: { display: true, text: title, font: { size: 16 } },
                        tooltip: {
                            callbacks: {
                                label: (context) => {
                                    let value = horizontal ? context.parsed.x : context.parsed.y;
                                    if (format === 'currency') return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value);
                                    if (format === 'percent') return value.toFixed(1) + '%';
                                    return value + (value === 1 ? ' ocorrência' : ' ocorrências');
                                }
                            }
                        }
                    },
                    scales: {
                        [horizontal ? 'x' : 'y']: {
                            beginAtZero: true,
                            ticks: {
                                callback: (value) => {
                                    if (format === 'currency') return value >= 1000 ? `R$ ${value / 1000}k` : `R$ ${value}`;
                                    if (format === 'percent') return value + '%';
                                    return Number.isInteger(value) ? value : null;
                                }
                            }
                        },
                        [horizontal ? 'y' : 'x']: { ticks: { autoSkip: false } }
                    }
                }
            };
            if (chartInstances[canvasId]) {
                chartInstances[canvasId].data.labels = labels;
                chartInstances[canvasId].data.datasets[0].data = data;
                chartInstances[canvasId].options.plugins.title.text = title; // Garante que o título seja atualizado
                chartInstances[canvasId].update();
            } else {
                chartInstances[canvasId] = new Chart(ctx, chartConfig);
            }
        }
    </script>
